  <!-- Pre-paint shell resolution, in dependency order: UI version â colour
       scheme â browser chrome colour. All three must land before first paint,
       and the last two read the first, so they share one block.

       UI version (docs/ecom/ui_switch/04-UI-SWITCH.md): ?ui= URL param (sticks
       via localStorage, for QA / dark-launch links) â VITE_UI_DEFAULT build env
       â the literal fallback below. That fallback is the TRACKED default:
       flipping it IS the flip, reverting it is the rollback. A no-commit
       emergency rollback instead sets VITE_UI_DEFAULT in the server's
       (gitignored) frontend/.env and rebuilds â a real env var outranks both
       the .env file and this fallback.

       The localStorage key carries a generation suffix. A QA pin must not
       outlive the flip it was made before: otherwise everyone who ever opened
       ?ui=â¦ or used the 7-tap toggle is frozen on that version, invisible to
       the default AND to a default rollback. Bump the suffix (here and in
       lib/uiVersion.js) whenever the tracked default changes. -->
  <script>
    (function () {
      var ui = 'v2';
      try {
        var q = new URLSearchParams(location.search).get('ui');
        if (q === 'v1' || q === 'v2') localStorage.setItem('ui_version_3', q);
        var d = '%VITE_UI_DEFAULT%';
        if (d.charAt(0) === '%') d = ''; // env var not defined at build time
        ui = localStorage.getItem('ui_version_3') || d || 'v2';
        ui = ui === 'v2' ? 'v2' : 'v1';
      } catch (e) {
        ui = 'v2'; // storage/URL unavailable â fall back to the SHIPPING UI
      }
      document.documentElement.dataset.ui = ui;

      // Colour scheme from the non-secret click-ctx marker (Â§5.1, M7).
      // Format: click-ctx=<present>.<theme>.<locale>. nginx can't stamp the
      // class into the content-hashed static shell, so we read the marker here;
      // ThemeContext (auto mode on Click) then only reconciles.
      // Suppressed under v2: it ships light-only until its dark tokens are
      // signed off (index.css's html[data-ui='v2'].dark is still a draft), and
      // ThemeContext forces the same â without this, Click on a dark SuperApp
      // would paint draft dark for a frame before being reconciled to light.
      var dark = false;
      if (ui !== 'v2') {
        var m = /(?:^|;\s*)click-ctx=[^.]*\.(dark|light)\./.exec(document.cookie);
        dark = !!m && m[1] === 'dark';
        if (dark) document.documentElement.classList.add('dark');
      }

      // Chrome colour (Android address bar, PWA splash + task switcher). The
      // static <meta> above can't follow data-ui, so it is restamped here: v1
      // keeps its navy untouched, v2 matches the surface its header actually
      // paints (index.css --surface). The dark value is dormant until v2 dark
      // ships, but stays correct so it needs no second edit then.
      var meta = document.querySelector('meta[name="theme-color"]');
      if (meta) {
        meta.setAttribute(
          'content',
          ui === 'v2' ? (dark ? '#1B222B' : '#FFFFFF') : '#0E2F5A',
        );
      }
    })();
  </script>
